cmake_minimum_required(VERSION 3.20)
project(NoiseDSP LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
# This directory is Desktop/Native; sources remain shared with SwiftPM.
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(NATIVE "${ROOT}/Sources/CNoiseRuntime")
add_library(noise_dsp SHARED ${NATIVE}/Denoiser.cpp ${NATIVE}/VoiceLeveler.cpp ${NATIVE}/MainSpeakerDenoiser.cpp)
target_include_directories(noise_dsp PRIVATE ${NATIVE}/include)
if(UNIX AND NOT APPLE)
  target_link_libraries(noise_dsp PRIVATE dl)
endif()
if(NOT DEFINED RNNOISE_SOURCE)
  message(FATAL_ERROR "RNNOISE_SOURCE must point to the pinned RNNoise source with downloaded model weights")
endif()
set(RN_FILES denoise rnn pitch kiss_fft celt_lpc nnet nnet_default parse_lpcnet_weights rnnoise_data rnnoise_tables)
list(TRANSFORM RN_FILES PREPEND "${RNNOISE_SOURCE}/src/")
list(TRANSFORM RN_FILES APPEND ".c")
add_library(rnnoise SHARED ${RN_FILES})
target_include_directories(rnnoise PRIVATE ${RNNOISE_SOURCE}/include ${RNNOISE_SOURCE}/src)
target_compile_definitions(rnnoise PRIVATE RNNOISE_BUILD _USE_MATH_DEFINES)
if(MSVC)
  target_compile_options(rnnoise PRIVATE /O2)
  target_compile_options(noise_dsp PRIVATE /O2 /EHsc)
else()
  target_compile_options(rnnoise PRIVATE -O3)
  target_compile_options(noise_dsp PRIVATE -O3)
endif()
install(TARGETS noise_dsp rnnoise RUNTIME DESTINATION . LIBRARY DESTINATION .)
